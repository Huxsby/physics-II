% DESCRIPCIÓN GENERAL:
% FABRIK es un método heurístico iterativo que resuelve el problema de cinemática
% inversa sin usar matrices de rotación ni ángulos explícitos. En su lugar, 
% encuentra cada posición de articulación localizando un punto en una línea.
% El algoritmo converge en pocas iteraciones, tiene bajo costo computacional
% y produce poses visualmente realistas.

% VENTAJAS PRINCIPALES:
% 1. No requiere matrices Jacobianas ni cálculos de derivadas
% 2. Evita singularidades matemáticas
% 3. Converge rápidamente (típicamente 2-4 iteraciones)
% 4. Fácil implementación y comprensión
% 5. Naturalmente incorpora restricciones
% 6. Produce movimientos suaves sin oscilaciones

% LIMITACIONES:
% 1. No garantiza solución única (puede haber múltiples configuraciones válidas)
% 2. No preserva automáticamente orientaciones de articulaciones intermedias
% 3. Requiere post-procesamiento para extraer ángulos de articulación

% ===============================================================================

Algorithm 1: FABRIK (Forward And Backward Reaching Inverse Kinematics)

Input: 
    - The joint positions pi for i = 1, . . . , n (where p1 is the base and pn is the end effector)
    - The target position t
    - The distances between each joint di = |pi+1 - pi| for i = 1,. . .,n - 1
    - Tolerance tol for convergence

Output: The new joint positions pi for i = 1, . . . , n.

% Check if the target is within reachable distance
dist_to_target = |pn - t|
total_length = sum(di) for i = 1 to n-1

if dist_to_target > total_length then
    % The target is unreachable; stretch the chain towards the target
    for i = 1 to n-1 do
        % Find the distance ri between the target t and the joint position pi
        ri = |t - pi|
        % Find the scaling factor ki to maintain link length
        ki = di/ri
        % Find the new joint positions pi+1 using linear interpolation
        pi+1 = (1 - ki) * pi + ki * t
    end for
else
    % The target is reachable; thus, set as b the initial position of the joint p1
    b = p1
    
    % Check whether the distance between the end effector pn and the target t is greater than a tolerance.
    difA = |pn - t|
    iteration = 0
    
    while difA > tol and iteration < max_iterations do
        
        % STAGE 1: FORWARD REACHING (from end effector to base)
        % Set the end effector pn as target t
        pn = t
        
        for i = n-1, n-2, . . . , 1 do
            % Calculate the distance between consecutive joints
            ri = |pi+1 - pi|
            % Calculate the unit direction vector from pi+1 to pi
            direction = (pi - pi+1) / ri
            % Place pi at distance di from pi+1 along the direction vector
            pi = pi+1 + di * direction
        end for
        
        % STAGE 2: BACKWARD REACHING (from base to end effector)
        % Set the root p1 to its initial position b
        p1 = b
        
        for i = 1, 2, . . . , n-1 do
            % Calculate the distance between consecutive joints
            ri = |pi+1 - pi|
            % Calculate the unit direction vector from pi to pi+1
            direction = (pi+1 - pi) / ri
            % Place pi+1 at distance di from pi along the direction vector
            pi+1 = pi + di * direction
        end for
        
        % Update the distance to target for convergence check
        difA = |pn - t|
        iteration = iteration + 1
        
    end while
end if



Algorithm 2: Joint Constraint Application (for restricted joints)

Input: 
    - The rotor R expressing the rotation between the orientation frames at joints pi and pi-1
    - Joint limits and constraint parameters for the joint
    - The current joint position pi

Output: The new re-oriented joint pi-1 that respects constraints.

% Este algoritmo maneja las restricciones de las articulaciones durante el proceso FABRIK
% Se aplica después de cada paso de Forward o Backward reaching para mantener 
% las restricciones físicas del robot

2.1 Check whether the rotor R is within the motion range bounds
    % Calcular el ángulo de rotación actual
    current_angle = extract_angle_from_rotor(R)
    
    % Verificar si está dentro de los límites permitidos
    if current_angle < joint_min_limit or current_angle > joint_max_limit then
        % Clamp the angle to the nearest boundary
        if current_angle < joint_min_limit then
            clamped_angle = joint_min_limit
        else
            clamped_angle = joint_max_limit
        end if
        
        % Crear nuevo rotor con el ángulo restringido
        R_constrained = create_rotor_from_angle(clamped_angle)
        
2.5 reorient the joint pi-1 in such a way that the rotor will be within the limits
        % Aplicar la transformación restringida
        % Recalcular la posición de pi-1 basada en la rotación restringida
        pi-1 = apply_constrained_rotation(pi, R_constrained, link_length)
    end if

% Nota: Este proceso se debe repetir para cada tipo de articulación

Algorithm 3: Target Constraint Application (for constrained targets)

Input: 
    - The target position t 
    - The angles defining the rotation constraints θj for j = 1,. . .,4
    - Joint position and orientation information

Output: The new target position t' that respects workspace constraints.

% Este algoritmo modifica el target para que esté dentro del workspace alcanzable
% considerando las restricciones de las articulaciones. Basado en secciones cónicas
% que definen el espacio de trabajo permitido.

3.1 Find the line equation L1
    % Definir la línea que conecta la base del robot con el target
    base_position = p1
    direction_vector = t - base_position
    L1 = {base_position + s * direction_vector | s ∈ ℝ}

3.2 Find the projection O of the target t on line L1
    % Proyectar el target sobre la línea L1 si es necesario
    s_projection = dot(t - base_position, direction_vector) / dot(direction_vector, direction_vector)
    O = base_position + s_projection * direction_vector

3.3 Find the distance between the point O and the joint position
    dist_O = |O - base_position|

3.4 Map the target (rotate and translate) to standard coordinate system
    % Transformar el problema a un sistema de coordenadas donde:
    % - O está en el origen
    % - Los ejes están alineados con los ejes de rotación principales
    transformed_target = transform_to_standard_frame(t, O, orientation_axes)

3.5 Solve the 2D simplified problem
    % Now it is a 2D simplified problem in the transformed space

3.6 Find in which quadrant the target belongs
    quadrant = determine_quadrant(transformed_target.x, transformed_target.y)

3.7 Find what conic section describes the allowed range of motion
    % Determinar la sección cónica (círculo, elipse, hipérbola, parábola) 
    % que describe el espacio de trabajo permitido en este cuadrante
    conic_type = determine_conic_type(joint_constraints, quadrant)

3.8 Find the conic section parameters for the quadrant
    % Find the conic section which is associated with that quadrant 
    % using the distances qj = S*tan(θj), where j = 1,..,4
    % S es un factor de escala basado en las longitudes de los eslabones
    S = calculate_scale_factor(link_lengths)
    for j = 1 to 4 do
        qj = S * tan(θj)
    end for
    
    % Definir los parámetros de la sección cónica
    conic_params = define_conic_section(qj, conic_type, quadrant)

3.9 Check whether the target is within the conic section or not
    if is_point_inside_conic(transformed_target, conic_params) then
        % use the true target position t
        t' = t
    else
        % Find the nearest point on that conic section from the target
        nearest_point = find_nearest_point_on_conic(transformed_target, conic_params)
        
        % Map (rotate and translate) that point back to original coordinate system
        % via reverse of step 3.4 and use that point as the new target position
        t' = transform_back_to_original_frame(nearest_point, O, orientation_axes)
    end if

% Notas importantes:
% - Este algoritmo es especialmente útil para robots con restricciones complejas
% - Las secciones cónicas proporcionan una representación matemática elegante 
%   del espacio de trabajo permitido
% - El algoritmo reduce el problema 3D a 2D para simplificar los cálculos
% - Se puede aplicar recursivamente para múltiples niveles de restricciones

Algorithm 4: Position to Joint Angles Conversion

Input:
    - Final joint positions pi for i = 1, . . . , n from FABRIK
    - Robot structure with link lengths and joint types
    - Screw axes Si for each joint

Output: Joint angles θi for i = 1, . . . , n-1

% Este algoritmo convierte las posiciones cartesianas obtenidas por FABRIK
% de vuelta a ángulos de articulación utilizables por el robot

4.1 Initialize transformation matrix
    T_accumulated = Identity_4x4_matrix
    θ = empty_array(n-1)

4.2 For each joint i from 1 to n-1 do
    % Calcular el vector actual del eslabón
    current_vector = pi+1 - pi
    
    % Obtener información del eje helicoidal
    Si = screw_axis[i]
    ωi = Si[1:3]  % Eje de rotación
    vi = Si[4:6]  % Parte lineal
    
    if joint_type[i] == "revolute" then
        % Para articulaciones de revolución
        % Calcular el vector de referencia en el sistema local
        T_inv = inverse(T_accumulated)
        local_current = transform_vector(T_inv, current_vector)
        
        % Proyectar vectores sobre el plano perpendicular al eje de rotación
        reference_direction = cross(ωi, joint_coordinates[i])
        projected_current = local_current - dot(local_current, ωi) * ωi
        projected_reference = reference_direction - dot(reference_direction, ωi) * ωi
        
        % Calcular ángulo entre vectores proyectados
        if |projected_current| > epsilon and |projected_reference| > epsilon then
            cos_θ = dot(projected_reference, projected_current) / (|projected_reference| * |projected_current|)
            sin_θ_sign = sign(dot(ωi, cross(projected_reference, projected_current)))
            θ[i] = arccos(clamp(cos_θ, -1, 1)) * sin_θ_sign
        else
            θ[i] = 0
        end if
        
    else if joint_type[i] == "prismatic" then
        % Para articulaciones prismáticas
        θ[i] = dot(current_vector, vi)
    end if
    
    % Actualizar transformación acumulada
    T_i = matrix_exponential(screw_matrix(Si, θ[i]))
    T_accumulated = T_accumulated * T_i

end for

Algorithm 5: Multi-Target FABRIK (for multiple end effectors)

Input:
    - Multiple target positions tj for j = 1, . . . , m
    - Shared joint positions and branching structure
    - Weights wj for each target importance

Output: Joint positions that satisfy all targets as best as possible

% Para robots con múltiples efectores finales (como manos con dedos)

5.1 Initialize all chains from shared base to each end effector
    for j = 1 to m do
        chain[j] = extract_kinematic_chain_to_target(j)
    end for

5.2 Main iteration loop
    iteration = 0
    while not_converged and iteration < max_iterations do
        
        % Forward reaching phase for each chain
        for j = 1 to m do
            % Apply FABRIK forward reaching to chain j
            apply_forward_reaching(chain[j], tj, wj)
        end for
        
        % Resolve conflicts in shared joints
        resolve_shared_joint_conflicts(chain, weights)
        
        % Backward reaching phase for each chain  
        for j = 1 to m do
            % Apply FABRIK backward reaching to chain j
            apply_backward_reaching(chain[j])
        end for
        
        % Check convergence for all targets
        converged = true
        for j = 1 to m do
            if |end_effector[j] - tj| > tolerance then
                converged = false
                break
            end if
        end for
        
        iteration = iteration + 1
    end while

Algorithm 6: FABRIK with Orientation Control

Input:
    - Target position t and target orientation R_target
    - Current joint positions and orientations
    - End effector offset and orientation constraint

Output: Joint positions and orientations satisfying both position and orientation

% Extensión de FABRIK para controlar también la orientación del efector final

6.1 Define virtual target for orientation control
    % Crear un target virtual desplazado del efector final
    offset_distance = small_positive_value
    virtual_target = t + R_target * [0, 0, offset_distance]

6.2 Apply standard FABRIK to position target
    % Usar Algorithm 1 para alcanzar la posición t
    apply_FABRIK_position(t)

6.3 Apply FABRIK to virtual target for orientation
    % Aplicar FABRIK adicional para el target virtual
    % Esto controla indirectamente la orientación
    end_effector_virtual = pn + current_orientation * [0, 0, offset_distance]
    
    % Aplicar corrección iterativa
    orientation_iterations = 0
    while orientation_error > orientation_tolerance and orientation_iterations < max_orientation_iterations do
        % Calcular error de orientación
        orientation_error = angular_distance(current_orientation, R_target)
        
        % Ajustar posiciones para mejorar orientación
        correction_vector = calculate_orientation_correction(R_target, current_orientation)
        virtual_target_corrected = virtual_target + correction_vector
        
        % Aplicar FABRIK hacia el target virtual corregido
        apply_FABRIK_position(virtual_target_corrected)
        
        % Recalcular orientación actual
        current_orientation = calculate_orientation_from_positions(pn-1, pn)
        orientation_iterations = orientation_iterations + 1
    end while
